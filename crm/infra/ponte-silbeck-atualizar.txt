bash <<'FIM'
# CRM Cabanas: atualiza a PONTE do Silbeck com os dados do Márcio (07/10/2026). Rodar no Cloud Shell.
# Servidor WireGuard do Márcio: 38.225.195.116:51820. CRM no túnel: 10.99.0.3/24.
# Passam pelo túnel: 10.99.0.0/24 e 192.168.132.0/24. Silbeck: 10.99.0.1:8366.
# A chave da ponte NÃO muda (fica guardada na máquina). Nenhum segredo aqui.
set -euo pipefail
gcloud config set project cabanas-crm >/dev/null
gcloud auth list --filter=status:ACTIVE --format='value(account)' | grep -q . || { echo "O Cloud Shell está sem login. Rode: gcloud auth login   (e depois cole este bloco de novo)"; exit 1; }
ZONA="southamerica-east1-a"; VM="silbeck-ponte"
PARTIDA="$(mktemp)"
cat > "$PARTIDA" <<'PARTIDA_FIM'
#!/bin/bash
# Roda a cada partida da máquina: instala o WireGuard, mantém a chave da ponte e monta o repasse ao Silbeck.
set -eu
MD=http://metadata.google.internal/computeMetadata/v1/instance
if ! command -v wg >/dev/null; then
  apt-get update -q && DEBIAN_FRONTEND=noninteractive apt-get install -y -q wireguard iptables unattended-upgrades
fi
umask 077; mkdir -p /etc/wireguard
[ -s /etc/wireguard/ponte.key ] || wg genkey > /etc/wireguard/ponte.key   # chave privada: nunca sai da máquina
PUB="$(wg pubkey < /etc/wireguard/ponte.key)"
at() { curl -sf -H 'Metadata-Flavor: Google' "$MD/attributes/$1"; }
HOTEL="$(at hotel-pubkey)"; FIMH="$(at hotel-endpoint || echo nenhum)"
TCRM="$(at tunel-crm || echo 10.99.0.3)"; THOTEL="$(at tunel-hotel || echo 10.99.0.1)"; DEST="$(at destino || echo 10.99.0.1)"
MASCARA="$(at tunel-mascara || echo 32)"   # o Márcio usa 10.99.0.3/24 (07/10/2026)
# Redes que passam pelo túnel ("redes", separadas por espaço; o Márcio: 10.99.0.0/24 192.168.132.0/24)
REDES="$(at redes || true)"
if [ -n "$REDES" ]; then PERMITIDOS="$(echo $REDES | sed 's/ /, /g')"; else PERMITIDOS="$THOTEL/32"; [ "$DEST" != "$THOTEL" ] && PERMITIDOS="$PERMITIDOS, $DEST/32"; fi
INTERNO="$(curl -sf -H 'Metadata-Flavor: Google' $MD/network-interfaces/0/ip)"
REGRAS="-t nat -A PREROUTING -d $INTERNO -p tcp --dport 8366 -j DNAT --to-destination $DEST:8366"
cat > /etc/wireguard/wg0.conf <<CONF
[Interface]
Address = $TCRM/$MASCARA
ListenPort = 51820
PrivateKey = $(cat /etc/wireguard/ponte.key)
PostUp = sysctl -qw net.ipv4.ip_forward=1; iptables $REGRAS; iptables -t nat -A POSTROUTING -o wg0 -j MASQUERADE; iptables -A FORWARD -i wg0 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT; iptables -A FORWARD -i wg0 -j DROP
PostDown = iptables ${REGRAS/-A PREROUTING/-D PREROUTING}; iptables -t nat -D POSTROUTING -o wg0 -j MASQUERADE; iptables -D FORWARD -i wg0 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT; iptables -D FORWARD -i wg0 -j DROP

[Peer]
# roteador MikroTik do hotel
PublicKey = $HOTEL
AllowedIPs = $PERMITIDOS
PersistentKeepalive = 25
CONF
[ "$FIMH" != "nenhum" ] && echo "Endpoint = $FIMH" >> /etc/wireguard/wg0.conf
systemctl enable wg-quick@wg0 >/dev/null 2>&1
systemctl restart wg-quick@wg0
curl -sf -X PUT --data "$PUB" -H 'Metadata-Flavor: Google' $MD/guest-attributes/ponte/pubkey
PARTIDA_FIM
echo "1/3 Gravando os dados do Márcio na ponte..."
gcloud compute instances add-metadata "$VM" --zone="$ZONA" \
  --metadata="hotel-pubkey=uiovxzVPkvUSLJJ/2cxA9dGnsq14uksUyo05ZcVNAiQ=,hotel-endpoint=38.225.195.116:51820,tunel-crm=10.99.0.3,tunel-mascara=24,tunel-hotel=10.99.0.1,destino=10.99.0.1,redes=10.99.0.0/24 192.168.132.0/24" \
  --metadata-from-file=startup-script="$PARTIDA" >/dev/null
rm -f "$PARTIDA"
echo "2/3 Reiniciando a ponte (1 a 2 minutos)..."
gcloud compute instances reset "$VM" --zone="$ZONA" --quiet >/dev/null
sleep 100
echo "3/3 Estado do túnel (o certo é aparecer 'latest handshake' e bytes recebidos):"
gcloud compute ssh "$VM" --zone="$ZONA" --tunnel-through-iap --quiet --command "sudo wg show; echo; sudo grep -E 'Address|AllowedIPs|Endpoint' /etc/wireguard/wg0.conf" 2>/dev/null \
  || echo "Não consegui entrar na ponte agora: espere 1 minuto e rode: gcloud compute ssh silbeck-ponte --zone=southamerica-east1-a --tunnel-through-iap --command 'sudo wg show'"
echo "Depois abra: https://crm-377803250649.southamerica-east1.run.app/saude/silbeck"
FIM
