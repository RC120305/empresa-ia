bash <<'FIM'
# CRM Cabanas: guarda no cofre do Google (Secret Manager) o endereço e as credenciais do Silbeck.
# Rodar no Cloud Shell. O script PERGUNTA cada valor: digite ali, nunca no chat.
# O client secret não aparece na tela enquanto você digita (é normal).
# Pode rodar de novo para trocar um valor (ex.: quando a Silbeck trocar o client secret).
set -euo pipefail
gcloud config set project cabanas-crm >/dev/null

guardar() { # nome valor
  if gcloud secrets describe "$1" >/dev/null 2>&1; then
    printf '%s' "$2" | gcloud secrets versions add "$1" --data-file=- >/dev/null
  else
    printf '%s' "$2" | gcloud secrets create "$1" --replication-policy=user-managed --locations=southamerica-east1 --data-file=- >/dev/null
  fi
  echo "  ✓ $1 guardado"
}

existe() { gcloud secrets describe "$1" >/dev/null 2>&1; }
echo "Endereço do Silbeck visto de fora do hotel. Deixe em branco (só Enter) o que já estiver certo no cofre."
while :; do
  read -r -p "IP público do hotel (ex.: 200.100.50.25): " HOST </dev/tty
  HOST="$(echo "$HOST" | tr -d '[:space:]')"
  if [ -z "$HOST" ] && existe silbeck-url; then break; fi
  echo "$HOST" | grep -Eq '^[A-Za-z0-9.-]+$' && break
  echo "  Isso não parece um IP (ex.: 200.100.50.25). Tente de novo."
done
while :; do
  read -r -p "Porta da API [8366]: " PORTA </dev/tty
  PORTA="${PORTA:-8366}"
  echo "$PORTA" | grep -Eq '^[0-9]{2,5}$' && break
  echo "  A porta é só número (ex.: 8366). Tente de novo."
done
read -r -p "client_id do Silbeck (Enter = manter o que já está no cofre): " CID </dev/tty
read -r -s -p "client_secret do Silbeck (não aparece ao digitar; Enter = manter): " CSEC </dev/tty; echo

[ -n "$HOST" ] && guardar silbeck-url "http://${HOST}:${PORTA}/datasnap/rest"
if [ -n "$CID" ]; then guardar silbeck-client-id "$CID"; elif ! existe silbeck-client-id; then echo "Faltou o client_id."; exit 1; fi
if [ -n "$CSEC" ]; then guardar silbeck-client-secret "$CSEC"; elif ! existe silbeck-client-secret; then echo "Faltou o client_secret."; exit 1; fi
unset CSEC CID

echo "----------------------------------------------------------------"
echo "Pronto! Agora abra: https://crm-377803250649.southamerica-east1.run.app/saude/silbeck"
echo "(o teste é refeito no máximo a cada 1 minuto)"
echo "----------------------------------------------------------------"
FIM
