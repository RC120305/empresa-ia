bash <<'FIM'
# CRM Cabanas: PONTE com o Silbeck por túnel WireGuard (opção A, 06/10/2026). Rodar no Cloud Shell.
# Cria uma máquina pequena na rede privada do CRM, ligada por WireGuard ao roteador MikroTik do hotel
# (07/10/2026: o Márcio configurou o WireGuard no MikroTik; endereços dele: CRM 10.99.0.3, MikroTik e Silbeck em 10.99.0.1). O CRM chama a máquina pelo endereço interno
# 10.20.0.250:8366 e ela repassa pelo túnel ao servidor do Silbeck na rede do hotel.
# Os dois lados se conectam: a ponte chama o MikroTik (se ele tiver endereço público) e o MikroTik chama a ponte.
# Pede só dados que não são segredo (chave PÚBLICA, IPs). A chave privada da ponte nunca sai da máquina.
# Pode rodar de novo (ex.: para trocar a chave pública do hotel): o que já existe é reaproveitado.
set -euo pipefail

PROJECT_ID="cabanas-crm"
REGION="southamerica-east1"
ZONA="southamerica-east1-a"
VM="silbeck-ponte"
IP_INTERNO="10.20.0.250"   # endereço da ponte dentro da rede do CRM (crm-sp, 10.20.0.0/24)
gcloud config set project "$PROJECT_ID" >/dev/null

ipok() { echo "$1" | grep -Eq '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; }
fora() { case "$1" in 10.20.0.*) echo "  A faixa 10.20.0.x já é a rede interna do CRM: peça ao Márcio uma faixa do túnel diferente (ex.: 10.99.0.x)."; return 1;; esac; }
gcloud auth list --filter=status:ACTIVE --format='value(account)' | grep -q . || { echo "O Cloud Shell está sem login. Rode: gcloud auth login   (e depois cole este script de novo)"; exit 1; }
while :; do
  read -r -p "1) Chave PÚBLICA do WireGuard do lado do hotel (servidor WireGuard do Márcio/MikroTik; termina com =): " HOTEL_PUB </dev/tty
  HOTEL_PUB="$(echo "$HOTEL_PUB" | tr -d '[:space:]')"
  echo "$HOTEL_PUB" | grep -Eq '^[A-Za-z0-9+/]{42}[AEIMQUYcgkosw480]=$' && break
  echo "  Isso não parece uma chave pública do WireGuard (44 caracteres, termina com =). Tente de novo."
done
while :; do
  read -r -p "2) Endereço público e porta desse WireGuard (ex.: 200.100.50.25:51820; Enter se não tiver): " HOTEL_END </dev/tty
  HOTEL_END="$(echo "$HOTEL_END" | tr -d '[:space:]')"
  [ -z "$HOTEL_END" ] && break
  echo "$HOTEL_END" | grep -Eq '^[A-Za-z0-9.-]+:[0-9]{2,5}$' && break
  echo "  Use IP:porta (ex.: 200.100.50.25:13231). Tente de novo."
done
while :; do
  read -r -p "3) IP do CRM dentro do túnel (o Márcio define) [10.99.0.3]: " TUN_CRM </dev/tty
  TUN_CRM="${TUN_CRM:-10.99.0.3}"; ipok "$TUN_CRM" && fora "$TUN_CRM" && break; ipok "$TUN_CRM" || echo "  Só o IP (ex.: 10.99.0.1)."
done
while :; do
  read -r -p "4) IP do servidor WireGuard do Márcio dentro do túnel [10.99.0.1]: " TUN_HOTEL </dev/tty
  TUN_HOTEL="${TUN_HOTEL:-10.99.0.1}"; ipok "$TUN_HOTEL" && fora "$TUN_HOTEL" && break; ipok "$TUN_HOTEL" || echo "  Só o IP (ex.: 10.99.0.2)."
done
while :; do
  read -r -p "5) IP para acessar o Silbeck pelo túnel (o Márcio repassa no MikroTik) [10.99.0.1]: " DESTINO </dev/tty
  DESTINO="${DESTINO:-10.99.0.1}"; ipok "$DESTINO" && break; echo "  Só o IP (ex.: 192.168.132.242)."
done
META="hotel-pubkey=${HOTEL_PUB},hotel-endpoint=${HOTEL_END:-nenhum},tunel-crm=${TUN_CRM},tunel-hotel=${TUN_HOTEL},destino=${DESTINO},enable-guest-attributes=TRUE"

echo "1/6 Endereços da ponte (interno fixo e público fixo)..."
gcloud compute addresses describe silbeck-ponte-interno --region="$REGION" >/dev/null 2>&1 || \
  gcloud compute addresses create silbeck-ponte-interno --region="$REGION" --subnet=crm-sp --addresses="$IP_INTERNO"
gcloud compute addresses describe silbeck-ponte-publico --region="$REGION" >/dev/null 2>&1 || \
  gcloud compute addresses create silbeck-ponte-publico --region="$REGION"
IP_PUBLICO="$(gcloud compute addresses describe silbeck-ponte-publico --region="$REGION" --format='value(address)')"

echo "2/6 Regras de firewall (só o necessário)..."
regra() { gcloud compute firewall-rules describe "$1" >/dev/null 2>&1 || gcloud compute firewall-rules create "$1" --network=crm-vpc --direction=INGRESS --target-tags=silbeck-ponte "${@:2}" >/dev/null; echo "  ✓ $1"; }
regra silbeck-ponte-wireguard --allow=udp:51820 --source-ranges=0.0.0.0/0          # túnel (protegido pela chave)
regra silbeck-ponte-crm       --allow=tcp:8366  --source-ranges=10.20.0.0/24       # só o CRM, pela rede interna
regra silbeck-ponte-ssh-iap   --allow=tcp:22    --source-ranges=35.235.240.0/20    # manutenção pelo console do Google

echo "3/6 Programa de partida da máquina..."
PARTIDA="$(mktemp)"
cat > "$PARTIDA" <<'PARTIDA_FIM'
#!/bin/bash
# Roda a cada partida da máquina: instala o WireGuard, mantém a chave da ponte e monta o repasse ao Silbeck.
set -eu
MD=http://metadata.google.internal/computeMetadata/v1/instance
if ! command -v wg >/dev/null; then
  apt-get update -q && DEBIAN_FRONTEND=noninteractive apt-get install -y -q wireguard iptables unattended-upgrades
fi
umask 077; mkdir -p /etc/wireguard
[ -s /etc/wireguard/ponte.key ] || wg genkey > /etc/wireguard/ponte.key   # chave privada: nunca sai da máquina
PUB="$(wg pubkey < /etc/wireguard/ponte.key)"
at() { curl -sf -H 'Metadata-Flavor: Google' "$MD/attributes/$1"; }
HOTEL="$(at hotel-pubkey)"; FIMH="$(at hotel-endpoint || echo nenhum)"
TCRM="$(at tunel-crm || echo 10.99.0.3)"; THOTEL="$(at tunel-hotel || echo 10.99.0.1)"; DEST="$(at destino || echo 10.99.0.1)"
PERMITIDOS="$THOTEL/32"; [ "$DEST" != "$THOTEL" ] && PERMITIDOS="$PERMITIDOS, $DEST/32"
INTERNO="$(curl -sf -H 'Metadata-Flavor: Google' $MD/network-interfaces/0/ip)"
REGRAS="-t nat -A PREROUTING -d $INTERNO -p tcp --dport 8366 -j DNAT --to-destination $DEST:8366"
cat > /etc/wireguard/wg0.conf <<CONF
[Interface]
Address = $TCRM/32
ListenPort = 51820
PrivateKey = $(cat /etc/wireguard/ponte.key)
PostUp = sysctl -qw net.ipv4.ip_forward=1; iptables $REGRAS; iptables -t nat -A POSTROUTING -o wg0 -j MASQUERADE; iptables -A FORWARD -i wg0 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT; iptables -A FORWARD -i wg0 -j DROP
PostDown = iptables ${REGRAS/-A PREROUTING/-D PREROUTING}; iptables -t nat -D POSTROUTING -o wg0 -j MASQUERADE; iptables -D FORWARD -i wg0 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT; iptables -D FORWARD -i wg0 -j DROP

[Peer]
# roteador MikroTik do hotel
PublicKey = $HOTEL
AllowedIPs = $PERMITIDOS
PersistentKeepalive = 25
CONF
[ "$FIMH" != "nenhum" ] && echo "Endpoint = $FIMH" >> /etc/wireguard/wg0.conf
systemctl enable wg-quick@wg0 >/dev/null 2>&1
systemctl restart wg-quick@wg0
curl -sf -X PUT --data "$PUB" -H 'Metadata-Flavor: Google' $MD/guest-attributes/ponte/pubkey
PARTIDA_FIM

echo "4/6 Máquina da ponte (e2-micro, Debian 12)..."
if gcloud compute instances describe "$VM" --zone="$ZONA" >/dev/null 2>&1; then
  gcloud compute instances add-metadata "$VM" --zone="$ZONA" --metadata="$META" --metadata-from-file=startup-script="$PARTIDA" >/dev/null
  gcloud compute instances reset "$VM" --zone="$ZONA" >/dev/null
else
  gcloud compute instances create "$VM" --zone="$ZONA" --machine-type=e2-micro \
    --image-family=debian-12 --image-project=debian-cloud --boot-disk-size=10GB \
    --network=crm-vpc --subnet=crm-sp --private-network-ip="$IP_INTERNO" --address="$IP_PUBLICO" \
    --can-ip-forward --tags=silbeck-ponte --no-service-account --no-scopes --shielded-secure-boot \
    --metadata="$META" --metadata-from-file=startup-script="$PARTIDA" >/dev/null
fi
rm -f "$PARTIDA"

echo "5/6 Esperando a máquina ligar o túnel (1 a 3 minutos)..."
PONTE_PUB=""
for i in $(seq 1 36); do
  PONTE_PUB="$(gcloud compute instances get-guest-attributes "$VM" --zone="$ZONA" --query-path=ponte/pubkey --format='value(value)' 2>/dev/null \
    || gcloud beta compute instances get-guest-attributes "$VM" --zone="$ZONA" --query-path=ponte/pubkey --format='value(value)' 2>/dev/null || true)"
  [ -n "$PONTE_PUB" ] && break; sleep 5
done
[ -n "$PONTE_PUB" ] || { echo "A máquina ainda não respondeu. Espere 2 minutos e rode o script de novo (ele reaproveita tudo)."; exit 1; }

echo "6/6 Endereço do Silbeck no cofre passa a ser a ponte..."
NOVA="http://${IP_INTERNO}:8366/datasnap/rest"
if [ "$(gcloud secrets versions access latest --secret=silbeck-url 2>/dev/null || true)" != "$NOVA" ]; then
  if gcloud secrets describe silbeck-url >/dev/null 2>&1; then
    printf '%s' "$NOVA" | gcloud secrets versions add silbeck-url --data-file=- >/dev/null
  else
    printf '%s' "$NOVA" | gcloud secrets create silbeck-url --replication-policy=user-managed --locations="$REGION" --data-file=- >/dev/null
  fi
fi
echo "  ✓ silbeck-url = $NOVA (o endereço antigo fica guardado como versão anterior)"

echo "----------------------------------------------------------------"
echo "Pronto! Passe ao Márcio estes dados (não são segredo):"
echo "  Chave pública do CRM (PublicKey) = ${PONTE_PUB}"
echo "  Endereço do CRM (Endpoint)       = ${IP_PUBLICO}:51820"
echo "  IP do CRM no túnel               = ${TUN_CRM}/32"
echo ""
echo "No MikroTik (RouterOS), algo como (troque WG pelo nome da interface WireGuard dele):"
echo "  /interface wireguard peers add interface=WG public-key=\"${PONTE_PUB}\" endpoint-address=${IP_PUBLICO} endpoint-port=51820 allowed-address=${TUN_CRM}/32 persistent-keepalive=25s comment=CRM"
echo "  /ip firewall filter add chain=forward in-interface=WG src-address=${TUN_CRM} dst-address=${DESTINO} protocol=tcp dst-port=8366 action=accept comment=CRM-Silbeck place-before=0"
echo "Se o servidor do Silbeck usar outro gateway, o Márcio faz src-nat (masquerade) do ${TUN_CRM} para a LAN."
echo "Quando o túnel estiver ativo, abra:"
echo "  https://crm-377803250649.southamerica-east1.run.app/saude/silbeck"
echo "(o teste é refeito no máximo a cada 1 minuto; o certo é aparecer \"tudo certo\")"
echo "----------------------------------------------------------------"
FIM
